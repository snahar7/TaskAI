onMutate: async ({ subtaskId, completed }) => {
  await queryClient.cancelQueries("/api/goals")
  await queryClient.cancelQueries([`/api/tasks/${parentTaskId}/subtasks`])
  await queryClient.cancelQueries([`/api/projects/${parentProjectId}/tasks`])
  // …then optimistically bump percentComplete on task & project in your cached data…
},
onSettled: () => {
  queryClient.invalidateQueries("/api/goals")
  queryClient.invalidateQueries([`/api/tasks/${parentTaskId}/subtasks`])
  queryClient.invalidateQueries([`/api/projects/${parentProjectId}/tasks`])
}
By doing both the backend cascade (so your database always has the correct percentages on subtask→task→project→goal) and the frontend cache invalidations (so your UI refetches the newly‐calculated values at every level), you’ll see the progress bars update immediately on subtasks and ripple upwards through tasks and projects.
